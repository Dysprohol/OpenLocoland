# Struct types, vtables, globals and function prototypes of Steam.exe for Ghidra.
# Applied by tools/ghidra/ApplyTypes.java after symbols.csv on every decompilation; the
# result is visible in re/c (fields instead of offsets) and re/c/types.h (all structs as C).
#
# Syntax (# starts a comment; a comment after a field becomes the field's description):
#
#   struct <Name> [<size>]          a struct; fields at explicit offsets, bytes between
#     <offset> <type> <name>        them stay unknown (types.h shows them as unknown_XXX)
#   end
#   vtable <Name> <address> <slots> a struct of function pointers built from the table in
#     <offset> <name>               the exe; slots not listed are named after the function
#   end                             in the slot (or vf_XX), typed from its prototype
#   global <address> <type> <name>  type of a global variable (name as in symbols.csv)
#   method <address> <Struct>       __thiscall with `this` typed <Struct>*; return value and
#                                   stack parameters stay as the decompiler finds them
#   func <address> <return> <convention> <Name>(<type> <name>, ...)
#                                   the whole prototype (name as in symbols.csv)
#   local <function> <stack offset> <type> <name>
#                                   a typed local variable (offset as in Ghidra's local_XX
#                                   names, e.g. -0x4c)
#
# Types: void char byte bool short ushort int uint float double undefined1/2/4, Windows
# types the program knows (RECT, HWND...), structs from this file; then any number of *
# and [n] (e.g. Vid*[0x800]). A struct embedded by value must be defined before it is used.

# ==== resource files (objects.res, .map, .vid, demos) ====================================

struct ResFile 0x40
  0x00 ResFile_vtbl* vtable
  0x04 uint flags                 # bit 0: a chunk is being written (closed by EndChunk)
  0x08 int lastAccess             # 0 write, 1 read, 2 seek - fseek is forced when switching
  0x0c char* name                 # file name (String)
  0x10 uint signature             # 'RES ' or 'RIFF'
  0x14 int chunkSize              # data size of the current chunk
  0x18 int chunkPos               # file position of the current chunk header
  0x1c int basePos                # file position of the container start
  0x20 int endPos                 # end of the container (size from the header + 8)
  0x24 uint chunkFlags            # bit 31: header with a sub-type; bit 8: rewritten
  0x28 int recordCount            # number of records in the current chunk
  0x2c int recordPos              # file position of the current record (its size field)
  0x30 int recordSize             # size of the current record
  0x34 int chunkSubType
  0x38 void* file                 # FILE*
  0x3c uint chunkId               # four-letter name of the current chunk
end

# ---- arithmetic coder of the compressed records (ResFile_ReadPacked / ResFile_WritePacked)
# The .vid frames with imageFlags 0x100 are stored arithmetic-coded: an ImageCodec object (0x2808
# bytes, vtable 0x47580c, base 0x475830 whose Encode/Decode slots are _purecall) keeps one
# RangeCoderModel per value of the previous byte, and codes 257 symbols (0-255 and 0x100 = the end
# of the record) with RangeCoder. Every model rescales by itself (RangeCoderModel_Update). The
# coder keeps `low` in a float-like form: bits 23-30 are the next byte to output, bits 0-22 the
# precision; the interval is 12 bits wide (1 << 12), the models sum to at most 2000. The 256 models
# are built with the CRT vector constructor iterator (ctor RangeCoderModel_Construct).

struct RangeCoder 0x18            # one coder state, used for encoding and for decoding
  0x00 uint low                   # low; bits 23-30 = the pending output byte, bits 0-22 = the precision
  0x04 uint range                 # 0x80000000 at the start, normalised to >= 0x800001
  0x08 uint pending               # encoding: the 0xff bytes waiting behind the pending byte (a carry); decoding: range >> bits
  0x0c byte buffer                # the pending output byte (encoding) / the last input byte (decoding)
  0x10 uint byteCount             # bytes written; the flush adds 5 and rounds with its half
  0x14 void* file                 # FILE*: output (encoding) or input (decoding)
end

struct RangeCoderModel 0x28       # the adaptive model of one byte context (257 symbols)
  0x00 int symbolCount            # 257 (0-255 and 0x100 = the end of the record)
  0x04 int symbolsToRescale       # countdown to the next rescale (set to totalFreq - the residue)
  0x08 int rescaleResidue         # the remainder carried into the next interval
  0x0c int totalFreq              # the current total frequency; doubles on every rescale up to maxTotal
  0x10 int maxTotal               # 2000
  0x14 int increment              # added to a symbol's frequency on each occurrence (the low short); +1 per rescale
  0x18 int shift                  # bits - 7 = 5: the log2 of the decoder index resolution
  0x1c ushort* cumFreqTable       # cumulative frequencies, symbolCount + 1 entries
  0x20 ushort* freqTable          # frequencies, symbolCount entries
  0x24 ushort* decoderTable       # cumFreq >> shift -> the first symbol of the range (0x102 bytes)
end

struct ImageCodec 0x2808          # arithmetic coder of a byte buffer (ResFile_ReadPacked/WritePacked)
  0x00 ImageCodec_vtbl* vtable    # 0x47580c (0x475830 while it is built): 0x18 Init, 0x1c Encode, 0x20 Decode
  0x04 int mode                   # 1 = the bytes in order, 2 = the even byte positions first and then the odd ones (.vid frame layout)
  0x08 RangeCoderModel[256] models # one model per value of the previous byte
end

# ---- script machine (.lgc): compiler 0x438810-0x43c6f0, interpreter Script_CallFunction
# 0x43c940, engine functions Script_CallExtern 0x43e5c0. Opcodes: DEVELOPERS.md "Script machine".
# Every script variable (globals, function parameters and function variables alike) is a
# fixed slot at the bottom of the value stack, numbered at compile time; the same stack holds
# the temporaries and the call frames above them. Functions therefore are not reentrant.

struct ScriptValue 0xc            # a stack slot / variable / temporary (ctor 0x43d8f0 int, 0x442460 object; dtor 0x401ae0)
  0x00 byte flags                 # 0x1 string (text), 0x2 int (value), 0x4 array (first element of `x[n]`), 0x8 has an initial value (written to demos and precompiled scripts), 0x10 object reference (unit "number"; cleared by Script_ObjectDeleted), 0x20 pointer (`int *p`: indexing goes through the slot number it holds), 0x40 not initialised
  0x04 int value                  # int, object pointer, or slot number (address) for & and arrays
  0x08 char* text                 # String; numbers in strings are converted with String_ToInt ("0x.." hex)
end

struct ScriptSymbol 0x18          # entry of the symbol table (copy 0x43e540, dtor 0x422de0)
  0x00 char* name                 # String, e.g. "main", "TrainNotAmmo", "F005_15", "F123_DESTROY", a variable or a label
  0x04 byte kind                  # 1 variable, 2 extern (engine function), 3 script function, 4 string constant, 5 int constant (4 and 5 are never created), 7 label, 8 label used by goto before its definition
  0x08 char* text                 # String: value of a string constant
  0x0c int value                  # variable: first slot; extern: function number (65-254, the opcode); function: code offset; label: code offset (kind 8: offset of the last jump operand waiting for it - a chain); int constant: the value
  0x10 int firstParam             # function / extern: slot of the first parameter (extern parameters hold the default values)
  0x14 int count                  # variable: number of elements (1, or n for x[n]); function / extern: number of parameters
end

struct ScriptDefine 0x8           # #define name value (dtor 0x422e50); expanded textually by Script_SkipSpace while compiling, dropped afterwards

struct Script 0x58                # the .lgc script machine, World+0x14c (one per level: <map name>.lgc and its #includes); released by Script_Release

# ---- Picture (wrapper, vtable 0x475410) and its implementation (vtable 0x4757a8)
# The wrapper holds only a pointer to the implementation (+4). Picture_Load picks the class from the
# file extension (.tga 1, .bmp 2, .flc 3, .z 5, unknown 0) and calls its Open slot. The implementation
# is an encoded file (a single picture or an animation) with a palette and one frame buffer; a
# trailing number in the name (name1.tga) means the next frame is in the next numbered file.
# Errors go to the log as "PICTURE '<name>'" with a code and a short text.
struct Picture 0xc             # the wrapper every caller holds (vtable 0x475410)
  0x000 Picture_vtbl* vtable
  0x004 PictureImpl* impl      # created by Picture_Construct, deleted by Picture_Delete
  0x008 int type               # the format the implementation was built for (Picture_Load, Picture_Create)
end

  0x00 Delete                  # 0x413180
  0x04 NextFrame               # 0x413160
  0x08 Rewind                  # 0x413170
  0x0c Load                    # 0x4294c0, also builds the object of the right class
end

struct PictureImpl 0x42c       # the picture/animation object of Picture_Construct (0x42c bytes)

  0x008 int frame              # the frame that is in the buffer (-1 before the first)
  0x00c int frameTime          # ms per frame (71 by default, the speed field of a FLIC)
  0x010 int width
  0x014 int height
  0x018 int bytesPerPixel      # 1 = palette, 2 = 16-bit 555, 3 = 24-bit, 4 = 32-bit
  0x01c char* name             # String with the file name (lower case)

# ---- rail network (R_MAP / R_DOT in the engine's log): Rail objects put points (RailDot) at
# both ends; points closer than 2 px share one RailDot. Trains move from point to point along
# the links; DEVELOPERS.md "Trains and rails".

struct RailLink 0x14              # one connection of a rail point to a neighbour
  0x00 RailDot* dot               # the other end
  0x04 int length                 # pixels: sqrt(dx^2 + (1.5 dy)^2) - y counts 1.5 times (isometric view)
  0x08 int backLink               # index of the reverse link in dot->links
  0x0c RailLink* crossing         # a link of another track crossing this one (RailMap_FindCrossings); a moving train on it blocks this link
  0x10 byte direction             # 0-255 from this point to the other
end

struct RailDot 0xec               # a rail point ("R_DOT"), created by RailMap_AddDot
  0x00 int refCount               # rails using the point; RailDot_Release deletes it at 0
  0x04 int broken                 # 1 while a rail here is destroyed; trains slow down to Options.brokenRailSpeed and repair the rails around when they pass the point
  0x0c int pushLink               # SetPushLine: the link trains must take from here (-1 none); entering a point against it is not allowed (RailDot_CanPass)
  0x10 int pushFlag               # SetPushLine push_flag
  0x14 int semaphore              # SetSemaphore value; 4+army on both ends closes the link for that army
  0x18 int linkCount              # at most 6 ("Too many links")
  0x1c RailLink[6] links
  0x94 int[6] searchDepth         # route search per incoming link (0xfffffff = not reached); [0] is also the depth of RailDot_Flood
  0xac int[6] searchCost
  0xc4 int[6] searchDistance
  0xdc Engine* occupant           # car standing on the point or on the link ahead (Engine_OccupyDots)
  0xe0 int x
  0xe4 int y
  0xe8 int z
end
